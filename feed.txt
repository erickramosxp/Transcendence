export async function getFeed(page = 0, size = 10) {

    const response = await api(`/posts?page=${page}&size=${size}`);
    return response;
}

export async function createPost(data) {
    const response = await api.post("/posts", data, {
        headers: {
            'Content-Type': 'application/json',
            'Accept': 'application/json'
        },
    });

    return response.data;
}

import { useState, useEffect } from "react";
import { extractUrl } from "@/shared/linkPreview";
import linkPreviewService from "@/shared/api/linkPreviewService";
import { useDebounce } from "@/hooks/useDebounce";

const LINK_PREVIEW_DEBOUNCE_MS = 600;

export function useCreatePost(onAddPost, onClose, postType) {
	const [text, setText] = useState("");
	const [file, setFile] = useState(null);
	const [showEmojiPicker, setShowEmojiPicker] = useState(false);
	const [linkPreview, setLinkPreview] = useState(null);
	const [linkPreviewLoading, setLinkPreviewLoading] = useState(false);
	const [linkPreviewError, setLinkPreviewError] = useState(null);

	const debouncedText = useDebounce(text, LINK_PREVIEW_DEBOUNCE_MS);

	useEffect(() => {
		const url = extractUrl(debouncedText);
		if (url) {
			setLinkPreviewLoading(true);
			setLinkPreviewError(null);
			linkPreviewService
				.fetchLinkPreview(url)
				.then((data) => {
					setLinkPreview(data);
					setLinkPreviewLoading(false);
					if (!data) {
						setLinkPreviewError("Preview indisponível para este link");
					}
				})
				.catch((error) => {
					setLinkPreviewError(error.message);
					setLinkPreviewLoading(false);
				});
		} else {
			setLinkPreview(null);
			setLinkPreviewLoading(false);
			setLinkPreviewError(null);
		}
	}, [debouncedText]);

	function handleSubmit() {
		if ( postType === "text" && text.trim() === "") return;

		if ((postType === "photo" || postType === "video") && file == null) return;

		onAddPost({
			text: text,
			file: file,
			fileType: typeof postType === "string" ? postType : postType?.type || "",
			linkPreview: linkPreview,
		});
		setText("");
		setFile(null);
		setLinkPreview(null);
		setLinkPreviewLoading(false);
		setLinkPreviewError(null);
		onClose();
	}

	function handleChange(e) {
		const value = e.target.value;
		setText(value);

		const match = value.match(/:(\w*)$/);
		if (match) {
			setShowEmojiPicker(true);
		} else {
			setShowEmojiPicker(false);
		}
	}

	function handleEmojiSelect(emojiData) {
		setText((prev) => prev.replace(/:(\w*)$/, "") + emojiData.emoji);
	}

	return {
		text,
		setText,
		linkPreview,
		linkPreviewLoading,
		linkPreviewError,
		file,
		setFile,
		showEmojiPicker,
		handleSubmit,
		handleChange,
		handleEmojiSelect,
	};
}

export default function CommentCard({ postId, incrementCommentCount }) {
    const [comments, setComments] = useState([]);
    const [commentText, setCommentText] = useState("");

    function handleSubmit(e) {
        e.preventDefault();
        const value = commentText.trim();
        if (!value) return;

        handleCreateComment();
    }

    const handleCreateComment = async () => {

        // trocar para definir no frontend primeiro e depois fazer a requisição
        try {

            const newComment = await createComment({postId: postId, text: commentText});

            setComments(prev => [...prev, newComment]);

            setCommentText("");
            incrementCommentCount(postId);

        } catch (err) {
            console.log(err);
        }
    };

    useEffect(() => {

        const requestComments = async () => {
            try {

                const data = await getComments(postId);
                const resp = data.data;
                setComments(resp.items);

            } catch (err) {
                console.log(err);
            }
        }

        requestComments();

    }, []);

    const handleToggleLike = async (commentId) => {

        try {

            const liked = comments.find(c => c.id == commentId).likedByMe;

            setComments(prev =>
                prev.map(c =>
                    c.id === commentId
                        ? {
                            ...c,
                            likedByMe: !c.likedByMe,
                            likeCount: c.likedByMe
                                ? c.likeCount - 1
                                : c.likeCount + 1
                        }
                        : c
                )
            );

            if (liked) {
                await unlikeComment(commentId);
            } else {
                await likeComment(commentId);
            }


        } catch (err) {
            console.error(err);
        }
    };

    return (
        <div className="pt-2 space-y-3">
            <Separator />
            {comments.length === 0 ? (
                <div className="text-sm text-gray-500 dark:text-gray-400">
                    Sem comentários
                </div>
            ) : (
                <Scroll className="max-h-50">
                    <div className="space-y-3">
                        {comments.map((comment) => (
                            <div key={comment.id ?? comment.user_id + comment.created_at} className="rounded-xl bg-slate-50 dark:bg-slate-800 p-3">
                                <div className="flex gap-3">
                                    <div>
                                        <Avatar
                                            src={comment.avatarUrl}
                                            username={comment.userName}
                                            name={comment.name}
                                            size="MD"
                                        />
                                    </div>
                                    <div className="flex-1">
                                        <div className="flex items-center gap-2 text-sm max-w-[200px] sm:max-w-none">
                                            <span className={`${yaleBlue.text.hover} truncate min-w-10 sm:truncate-none`}>
                                                <Link className='truncate' to={`/${comment.userName}`}>
                                                    {comment.name}
                                                </Link>
                                            </span>

                                            <span className="text-slate-500 truncate sm:truncate-none">
                                                @{comment.userName}
                                            </span>

                                            <span title={new Date(comment.created_at).toLocaleString()} className="text-slate-400 truncate sm:truncate-none">
                                                {getCurrentTime(comment.created_at)}
                                            </span>
                                        </div>
                                        <div className="mt-2 leading-relaxed">
                                            <TextWithLink text={comment.contents} />
                                        </div>
                                        <div className="flex items-center gap-6 mt-3 text-slate-500">
                                            <button
                                                onClick={() => handleToggleLike(comment.id)}
                                                className={`flex items-center gap-2 text-sm transition-all hover:scale-105 ${comment.likedByMe
                                                    ? 'text-pink-600 dark:text-pink-400'
                                                    : 'text-slate-500 dark:text-slate-400 hover:text-pink-600 dark:hover:text-pink-400'
                                                    }`}
                                            >
                                                <Heart className={`h-5 w-5 ${comment.likedByMe ? 'fill-current' : ''}`} />
                                                {comment.likeCount}
                                            </button>

                                        </div>
                                    </div>
                                </div>
                            </div>
                        ))}
                    </div>
                </Scroll>
            )}
            <form onSubmit={handleSubmit} className="space-y-2">
                <div className="flex items-center justify-end gap-2">
                    <Textarea
                        maxLength={500}
                        value={commentText}
                        onChange={(e) => setCommentText(e.target.value)}
                        onKeyDown={(e) => {
                            if (e.key === "Enter" && !e.shiftKey) {
                                e.preventDefault();
                                handleSubmit(e);
                            }
                        }}
                        placeholder="Escreva um comentário..."
                        className="min-h-[40px] resize-none"
                    />
                    <div>
                        <Button
                            onClick={handleSubmit}
                            size="sm"
                        >
                            <Send className="h-4 w-4" />
                        </Button>
                    </div>
                </div>
            </form>
        </div>
    );
}

function PostList({posts, setPosts}) {

    const [activeCommentsPostId, setActiveCommentsPostId] = useState(null);
    const [isLoadingMore, setIsLoadingMore] = useState(false);
    const [isLoadingInitialFeed, setLoadingFeed] = useState(true);
    const [page, setPage] = useState(0);
    const [hasMore, setHasMore] = useState(false);
    const sentinelRef = useRef(null);

    useEffect(() => {

        async function loadInitialFeed() {
            try {
                const data = await getFeed(0, 10);
                const resp = data.data;
                setPosts(resp.items);
                setPage(resp.page + 1);
                setHasMore(resp.hasMore);
            } catch (err) {
                console.error(err);
            } finally {
                setLoadingFeed(false);
            }
        }
        
        loadInitialFeed();
    }, []);

    const loadMore = useCallback(async () => {

        if (isLoadingMore || !hasMore) return;

        setIsLoadingMore(true);

        try {
            const data = await getFeed(page, 10);
            const items = data.items || data.data?.items || [];

            if (items.length === 0) {
                setHasMore(false);
                return ;
            }
            setPosts(prev => [...prev, ...items]);
            setPage(data.page + 1);
            setHasMore(data.hasMore);

        } catch (err) {
            console.error(err);
            toast.error('Falha ao carregar os posts');
        } finally {
            setIsLoadingMore(false);
        }

    }, [hasMore, isLoadingMore, page]);

    useInfiniteScroll(sentinelRef, loadMore, {
        enabled: hasMore && !isLoadingMore,
        rootMargin: "200px",
    });


//     {
//   "items": [
//     {
//       "id": 0,
//       "name": "string",
//       "userName": "string",
//       "contents": "string",
//       "avatarUrl": "string",
//       "likedByMe": true,
//       "likeCount": 0,
//       "commentCount": 0,
//       "media": "string",
//       "created_at": "2022-03-10T12:15:50-04:00"
//     }
//   ],
//   "page": 0,
//   "size": 0,
//   "hasMore": true
// }

    async function handleLike(postId) {

        const liked = posts.find(p => p.id == postId).likedByMe;

        setPosts(prev =>
                prev.map(p =>
                    p.id === postId
                        ? {
                            ...p,
                            likedByMe: !p.likedByMe,
                            likeCount: p.likedByMe
                                ? p.likeCount - 1
                                : p.likeCount + 1
                        }
                        : p
                )
        );

        if (liked) {
            await unlikePost(postId);
        } else {
            await likePost(postId);
        }
    }


    function toggleComments(postId) {
        console.log("Toggle comments for post", postId);
        setActiveCommentsPostId((prev) => (prev === postId ? null : postId));
    }

	const incrementCommentCount = (postId) => {

		setPosts(prev => 
			prev.map( p => p.id === postId ?
				{ ...p, commentCount: (p.commentCount ?? 0) + 1 }
                : p
			)

		)
	}

    if(isLoadingInitialFeed) {
        return (
            <Loading  className="py-10"/>
        );
    }

	return (
		<div className="space-y-8">
			{posts && posts.length === 0 ? (
				<div className="text-center text-gray-500 dark:text-gray-400 py-8">
					Nada por aqui ainda.<br/>
					Seja o primeiro a compartilhar algo ✨
				</div>
			) : (
				<>
					{posts && posts.map((post) => (
						<Card
							key={post.id}
							className=" border-slate-200/50 dark:border-slate-800/50 shadow-xl bg-white/80 dark:bg-slate-900/80 backdrop-blur-xl"
						>
							<PostCard
								post={post}
								onLike={handleLike}
								onToggleComments={toggleComments}
								activeCommentsPostId={activeCommentsPostId}
								incrementCommentCount={incrementCommentCount}
							/>
						</Card>
					))}
					<div ref={sentinelRef} className="h-4" aria-hidden />
					{isLoadingMore && (
						<div className="text-center text-slate-500 dark:text-slate-400 py-4">
							Carregando mais…
						</div>
					)}
					{!hasMore && (
						<div className="text-center text-slate-400 dark:text-slate-500 py-4 text-sm">
							Você chegou ao fim do feed
						</div>
					)}
				</>
			)}
		</div>
	);
}

export default PostList;
